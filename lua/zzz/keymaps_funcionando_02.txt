local function InsertHeader()
  local buf = 0
  local filepath = vim.api.nvim_buf_get_name(buf):gsub("^" .. vim.fn.expand("~"), "~")

  local ft_comments = {
    lua = "--",
    sh = "#",
    bash = "#",
    typescript = "//",
    javascript = "//",
    go = "//",
  }

  local comment = ft_comments[vim.bo.filetype] or "--"
  vim.api.nvim_buf_set_lines(buf, 0, 0, false, {
    comment .. " @file: " .. filepath,
    comment .. " @mission: ",
    "",
  })
end
